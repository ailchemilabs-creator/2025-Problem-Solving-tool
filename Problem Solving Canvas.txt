// --- REUSABLE COMPONENTS ---
const DictationInput = ({ value, onChange, placeholder, className = "", multiline = false, autoFocus = false }) => {
  const [isListening, setIsListening] = useState(false);
  const [supported, setSupported] = useState(true);
  const recognitionRef = useRef(null);

  useEffect(() => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
      setSupported(false);
      return;
    }
    const recognition = new SpeechRecognition();
    recognition.continuous = false;
    recognition.interimResults = true;
    
    recognition.onstart = () => setIsListening(true);
    recognition.onend = () => setIsListening(false);
    
    recognition.onresult = (event) => {
      let finalTranscript = '';
      for (let i = event.resultIndex; i < event.results.length; ++i) {
        if (event.results[i].isFinal) finalTranscript += event.results[i][0].transcript;
      }
      if (finalTranscript) onChange(value ? `${value} ${finalTranscript}` : finalTranscript);
    };
    recognitionRef.current = recognition;
  }, [value, onChange]);

  const toggleListening = () => {
    if (isListening) recognitionRef.current?.stop();
    else try { recognitionRef.current?.start(); } catch (e) { console.error(e); }
  };

  const Component = multiline ? 'textarea' : 'input';
  
  return (
    <div className={`relative w-full group ${className}`}>
      <Component
        autoFocus={autoFocus}
        value={value || ''}
        onChange={(e) => onChange(e.target.value)}
        placeholder={placeholder}
        className={`w-full bg-white dark:bg-slate-900 border-2 border-slate-200 dark:border-slate-700 text-slate-800 dark:text-slate-100 rounded-xl px-4 py-3 focus:outline-none focus:border-black dark:focus:border-slate-400 focus:ring-4 focus:ring-slate-100 dark:focus:ring-slate-800 transition-all font-medium ${multiline ? 'min-h-[100px] resize-y' : ''} pr-12`}
      />
      {supported && (
        <button
          onClick={toggleListening}
          className={`absolute right-3 ${multiline ? 'top-3' : 'top-1/2 -translate-y-1/2'} p-2 rounded-lg transition-colors ${isListening ? 'bg-red-100 dark:bg-red-900/50 text-red-600 dark:text-red-400 animate-pulse' : 'text-slate-400 dark:text-slate-500 hover:bg-slate-100 dark:hover:bg-slate-800 hover:text-black dark:hover:text-white lg:opacity-0 lg:group-hover:opacity-100 focus:opacity-100'}`}
          title="Dictate"
        >
          {isListening ? <Icons.Mic className="w-5 h-5" /> : <Icons.MicOff className="w-5 h-5" />}
        </button>
      )}
    </div>
  );
};

const SidebarItem = ({ p, active, onOpen, onTogglePin, onDelete, onRename }) => {

   return (
      <div 
        draggable 
        onDragStart={e => e.dataTransfer.setData('problemId', p.id)}
        onClick={() => !isEditing && onOpen(p.id)}
        className={`group flex items-center justify-between p-2 rounded-lg cursor-pointer transition-colors ${active ? 'bg-blue-50 dark:bg-blue-900/30 text-blue-700 dark:text-blue-300' : 'hover:bg-slate-100 dark:hover:bg-slate-800 text-slate-700 dark:text-slate-300'}`}

       setActiveProblemId(id);

'bg-blue-50 dark:bg-blue-900/30 text-blue-700 dark:text-blue-300' : 'bg-transparent hover:bg-slate-100 dark:hover:bg-slate-800 text-slate-700 dark:text-slate-300'}`}>

const ProblemSession = ({ problem, updateMemory, setSidebarOpen, isDarkMode, toggleTheme }) => {
  const [currentIndex, setCurrentIndex] = useState(0);
  const SLIDES = [Slide1, Slide2, Slide3, Slide4, Slide5, Slide6, Slide7, Slide8, Slide9, Slide10];

  // Reset to the first slide whenever a different problem is opened
  useEffect(() => {
    setCurrentIndex(0);
  }, [problem.id]);

  const nextSlide = () => { if (currentIndex < SLIDES.length - 1) setCurrentIndex(prev => prev + 1); };
  const prevSlide = () => { if (currentIndex > 0) setCurrentIndex(prev => prev - 1); };

  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA' || e.target.tagName === 'SELECT') return;
      if (e.key === 'ArrowRight') nextSlide();
      if (e.key === 'ArrowLeft') prevSlide();
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [currentIndex]);

  const touchStartX = useRef(null);
  const touchEndX = useRef(null);
  const minSwipeDistance = 50;

  const onTouchStart = (e) => { touchEndX.current = null; touchStartX.current = e.targetTouches[0].clientX; };
  const onTouchMove = (e) => { touchEndX.current = e.targetTouches[0].clientX; };
  const onTouchEnd = (e) => {
    if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA' || e.target.tagName === 'SELECT') return;
    if (!touchStartX.current || !touchEndX.current) return;
    const distance = touchStartX.current - touchEndX.current;
    if (distance > minSwipeDistance) nextSlide();
    if (distance < -minSwipeDistance) prevSlide();
  };

  return (
    <div className="h-full w-full flex flex-col relative" onTouchStart={onTouchStart} onTouchMove={onTouchMove} onTouchEnd={onTouchEnd}>
      
      <style dangerouslySetInnerHTML={{__html: `
        .clip-polygon-bottom { clip-path: polygon(0% 0%, 100% 0%, 100% 70%, 50% 100%, 0% 70%); }
        .clip-polygon-bottom-large { clip-path: polygon(0% 0%, 100% 0%, 100% 60%, 50% 100%, 0% 60%); }
        .custom-scrollbar::-webkit-scrollbar { width: 6px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background-color: rgba(156, 163, 175, 0.5); border-radius: 10px; }
        .dark { color-scheme: dark; }
      `}} />

      {/* TOP PROGRESS BAR */}
      <div className="absolute top-0 left-0 right-0 h-1.5 sm:h-2 bg-slate-200 dark:bg-slate-800 z-50 flex">
         {SLIDES.map((_, i) => <div key={i} className={`flex-1 transition-all duration-500 ${i <= currentIndex ? 'bg-black dark:bg-white' : ''}`} />)}
      </div>

      {/* HEADER OVERLAY WITH HAMBURGER */}
      <div className="absolute top-4 sm:top-6 left-4 sm:left-6 right-4 sm:right-6 flex items-center justify-between z-50 pointer-events-none">
         <button onClick={() => setSidebarOpen(true)} className="pointer-events-auto p-2 sm:px-4 sm:py-2.5 bg-white/80 dark:bg-slate-900/80 backdrop-blur shadow-sm border border-slate-200 dark:border-slate-700 rounded-xl sm:rounded-full font-bold text-xs sm:text-sm text-slate-700 dark:text-slate-300 hover:bg-black hover:text-white dark:hover:bg-white dark:hover:text-black transition-colors flex items-center gap-2">
            <Icons.Menu className="w-5 h-5 sm:w-4 sm:h-4" /> <span className="hidden sm:inline">Menu</span>
         </button>

         <div className="flex items-center gap-3 pointer-events-auto">
            <div className="hidden sm:block text-[10px] sm:text-xs font-bold tracking-widest text-slate-400 dark:text-slate-500 select-none bg-white/80 dark:bg-slate-900/80 px-4 py-2 rounded-full shadow-sm border border-slate-200 dark:border-slate-700">
               USE ← → KEYS TO NAVIGATE
            </div>
            <button onClick={toggleTheme} className="p-2 sm:p-2.5 rounded-xl sm:rounded-full bg-white/80 dark:bg-slate-900/80 text-slate-600 dark:text-slate-300 shadow-sm border border-slate-200 dark:border-slate-700 hover:bg-white dark:hover:bg-slate-800 transition-all active:scale-90">
              {isDarkMode ? <Icons.Sun className="w-5 h-5" /> : <Icons.Moon className="w-5 h-5" />}
            </button>
         </div>
      </div>

      {/* HORIZONTAL SCROLLING CANVAS */}
      <div className="flex-1 w-full h-full overflow-hidden relative">
        <div 
          className="flex h-full w-full transition-transform duration-500 ease-in-out" 
          style={{ transform: `translateX(-${currentIndex * 100}%)` }}
        >
          {SLIDES.map((SlideComponent, index) => (
            <div key={index} className="w-full h-full flex-shrink-0 p-4 sm:p-12 pb-40 overflow-y-auto overflow-x-hidden pt-20 sm:pt-24 custom-scrollbar">
               <SlideComponent memory={problem.memory} update={updateMemory} />
            </div>
          ))}
        </div>
      </div>

      {/* FLOATING NAVIGATION CONTROLS */}
      <div className="absolute bottom-6 sm:bottom-8 left-0 right-0 pointer-events-none flex justify-center z-50 px-4">
        <div className="bg-white/95 dark:bg-slate-900/95 backdrop-blur-md shadow-[0_10px_40px_rgb(0,0,0,0.15)] dark:shadow-[0_10px_40px_rgba(0,0,0,0.4)] border border-slate-200 dark:border-slate-700 rounded-full p-2 flex items-center justify-between pointer-events-auto max-w-[400px] w-full transition-colors duration-500">
          <button onClick={prevSlide} disabled={currentIndex === 0} className="p-3 sm:p-4 bg-slate-100 dark:bg-slate-800 hover:bg-slate-200 dark:hover:bg-slate-700 text-slate-800 dark:text-slate-200 rounded-full disabled:opacity-30 transition-all active:scale-95 flex items-center gap-2">
            <Icons.ArrowLeft className="w-5 h-5 sm:w-6 sm:h-6" /> <span className="text-xs font-bold hidden sm:inline pr-2">PREV</span>
          </button>
          <div className="font-bold text-xs sm:text-sm tracking-widest text-slate-500 dark:text-slate-400 text-center select-none whitespace-nowrap">
            PAGE {currentIndex + 1} OF {SLIDES.length}
          </div>
          <button onClick={nextSlide} disabled={currentIndex === SLIDES.length - 1} className="p-3 sm:p-4 bg-black dark:bg-white hover:bg-slate-800 dark:hover:bg-slate-200 text-white dark:text-black rounded-full disabled:opacity-30 transition-all active:scale-95 flex items-center gap-2">
            <span className="text-xs font-bold hidden sm:inline pl-2">NEXT</span> <Icons.ArrowRight className="w-5 h-5 sm:w-6 sm:h-6" />
          </button>
        </div>
      </div>
    </div>
  );
};

export default function App() {
  const [folders, setFolders] = useState([]);
  const [problems, setProblems] = useState([]);
  const [activeProblemId, setActiveProblemId] = useState(null);
  const [isDarkMode, setIsDarkMode] = useState(false);
  const [isSidebarOpen, setIsSidebarOpen] = useState(false);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    try {
      const savedFolders = localStorage.getItem('ps_folders');
      const savedProblems = localStorage.getItem('ps_problems');
      const savedTheme = localStorage.getItem('ps_theme');
      
      if (savedFolders) setFolders(JSON.parse(savedFolders));
      if (savedProblems) setProblems(JSON.parse(savedProblems));
      if (savedTheme) setIsDarkMode(savedTheme === 'dark');
    } catch (e) { console.error(e); }
    setLoaded(true);
  }, []);

  useEffect(() => {
    if (!loaded) return;
    localStorage.setItem('ps_folders', JSON.stringify(folders));
    localStorage.setItem('ps_problems', JSON.stringify(problems));
    localStorage.setItem('ps_theme', isDarkMode ? 'dark' : 'light');
  }, [folders, problems, isDarkMode, loaded]);

  const toggleTheme = () => setIsDarkMode(!isDarkMode);

  const createProblem = () => {
    const newProblem = { id: 'p' + Date.now(), title: `Problem ${new Date().toLocaleDateString()}`, folderId: null, memory: {}, updatedAt: Date.now(), isPinned: false };
    setProblems([newProblem, ...problems]);
    setActiveProblemId(newProblem.id);
    if(window.innerWidth < 768) setIsSidebarOpen(false);
  };

  const createFolder = () => {
    const newFolder = { id: 'f' + Date.now(), name: 'New Folder', color: FOLDER_COLORS[0] };
    setFolders([newFolder, ...folders]);
  };

  return (
    <div className={`h-[100dvh] w-screen overflow-hidden flex font-sans transition-colors duration-500 ${isDarkMode ? 'dark bg-zinc-950 text-slate-100' : 'bg-zinc-50 text-slate-800'}`}>
       <Sidebar 
         folders={folders} problems={problems} activeProblemId={activeProblemId} setActiveProblemId={setActiveProblemId} 
         setFolders={setFolders} setProblems={setProblems} isSidebarOpen={isSidebarOpen} setSidebarOpen={setIsSidebarOpen} 
         createProblem={createProblem} createFolder={createFolder}
       />
       
       <div className="flex-1 relative overflow-hidden flex flex-col h-full w-full">
          {activeProblemId ? (
             <ProblemSession 
                problem={problems.find(p => p.id === activeProblemId)} 
                updateMemory={(key, val) => setProblems(prev => prev.map(p => p.id === activeProblemId ? { ...p, memory: { ...p.memory, [key]: val }, updatedAt: Date.now() } : p))} 
                setSidebarOpen={setIsSidebarOpen} 
                isDarkMode={isDarkMode} 
                toggleTheme={toggleTheme}
             />
          ) : (
             <Home createProblem={createProblem} setSidebarOpen={setIsSidebarOpen} isDarkMode={isDarkMode} toggleTheme={toggleTheme} />
          )}
          
          {isSidebarOpen && <div className="absolute inset-0 bg-black/20 dark:bg-black/50 z-50 md:hidden backdrop-blur-sm" onClick={() => setIsSidebarOpen(false)} />}
       </div>
    </div>
  );
}
